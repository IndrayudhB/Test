script.js 
// Store the current expression as a string
let expression = '';

// Read the display elements from the page
const resultDisplay     = document.getElementById('result');
const expressionDisplay = document.getElementById('expression');
const feedback          = document.getElementById('feedback');

// Append a value (digit or operator) to the expression
function appendValue(value) {
  clearFeedback();

  // Prevent starting expression with an operator (except minus for negative)
  if (expression === '' && ['+', '*', '/'].includes(value)) {
    showFeedback('Expression cannot start with an operator.');
    return;
  }

  // Prevent two consecutive operators
  const lastChar = expression.slice(-1);
  const operators = ['+', '-', '*', '/'];
  if (operators.includes(lastChar) && operators.includes(value)) {
    showFeedback('Cannot enter two operators in a row.');
    return;
  }

  // Prevent multiple decimal points in the same number
  if (value === '.') {
    const parts = expression.split(/[\+\-\*\/]/);
    const currentPart = parts[parts.length - 1];
    if (currentPart.includes('.')) {
      showFeedback('Number already contains a decimal point.');
      return;
    }
  }

  // Append the value and update both display lines
  expression += value;
  expressionDisplay.textContent = formatExpression(expression);
  resultDisplay.textContent = expression;
}

// Replace raw operators with readable symbols for the expression line
function formatExpression(expr) {
  return expr
    .replace(/\*/g, ' × ')
    .replace(/\//g, ' ÷ ')
    .replace(/\+/g, ' + ')
    .replace(/-/g, ' − ');
}

// Delete the last character from the expression
function deleteLast() {
  clearFeedback();
  if (expression === '') return;
  expression = expression.slice(0, -1);
  expressionDisplay.textContent = expression ? formatExpression(expression) : '\u00A0';
  resultDisplay.textContent     = expression || '0';
}

// Clear everything and reset to zero
function clearAll() {
  expression = '';
  expressionDisplay.textContent = '\u00A0';
  resultDisplay.textContent = '0';
  resultDisplay.style.color = '';
  clearFeedback();
}

// Evaluate the expression and display the result
function calculate() {
  clearFeedback();

  // Validation: empty input check
  if (expression.trim() === '') {
    showFeedback('Please enter a number or expression first.');
    return;
  }

  // Validation: expression must not end with an operator
  const lastChar = expression.slice(-1);
  if (['+', '-', '*', '/'].includes(lastChar)) {
    showFeedback('Expression is incomplete. Please enter a number after the operator.');
    return;
  }

  try {
    // Evaluate the expression safely
    const answer = Function('"use strict"; return (' + expression + ')')();

    // Handle division by zero
    if (!isFinite(answer)) {
      showFeedback('Cannot divide by zero.');
      resultDisplay.style.color = 'var(--red)';
      return;
    }

    // Round to avoid floating point display issues
    const rounded = parseFloat(answer.toFixed(10));

    // Dynamically update the page with the result
    expressionDisplay.textContent = formatExpression(expression) + ' =';
    resultDisplay.textContent     = rounded;
    resultDisplay.style.color     = 'var(--green)';

    // Set expression to result so user can continue calculating
    expression = String(rounded);

  } catch (e) {
    showFeedback('Invalid expression. Please check your input.');
    resultDisplay.style.color = 'var(--red)';
  }
}

// Show a validation feedback message
function showFeedback(message) {
  feedback.textContent = message;
}

// Clear the feedback message
function clearFeedback() {
  feedback.textContent = '\u00A0';
  resultDisplay.style.color = '';
}

// Allow keyboard input for convenience
document.addEventListener('keydown', function(e) {
  if (e.key >= '0' && e.key <= '9') appendValue(e.key);
  else if (e.key === '+') appendValue('+');
  else if (e.key === '-') appendValue('-');
  else if (e.key === '*') appendValue('*');
  else if (e.key === '/') { e.preventDefault(); appendValue('/'); }
  else if (e.key === '.') appendValue('.');
  else if (e.key === 'Enter' || e.key === '=') calculate();
  else if (e.key === 'Backspace') deleteLast();
  else if (e.key === 'Escape') clearAll();
});
