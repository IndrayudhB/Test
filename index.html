ATM Transaction

Custom exception · InsufficientFundsException · IllegalArgumentException
Exception Handling Easy

import java.util.Scanner;

// Custom exception for insufficient funds
class InsufficientFundsException extends Exception {
    public InsufficientFundsException(String message) {
        super(message);
    }
}

public class Solution {

    public static void withdraw(int balance, int amount)
            throws InsufficientFundsException {

        if (amount <= 0) {
            throw new IllegalArgumentException(
                "Error: Withdrawal amount must be greater than zero.");
        }
        if (amount > balance) {
            throw new InsufficientFundsException(
                "Error: Insufficient funds in your account. Your current balance is "
                + balance + ".");
        }
        System.out.println("Transaction successful. Remaining balance: "
            + (balance - amount) + ".");
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int balance = sc.nextInt();
        int amount  = sc.nextInt();
        try {
            withdraw(balance, amount);
        } catch (InsufficientFundsException e) {
            System.out.println(e.getMessage());
        } catch (IllegalArgumentException e) {
            System.out.println(e.getMessage());
        }
        sc.close();
    }
}

Key Concepts

  • Custom exception InsufficientFundsException extends Exception
  • Check amount ≤ 0 first → throw IllegalArgumentException
  • Check amount > balance → throw InsufficientFundsException
  • Print remaining balance on success

Order of checks

Always validate the amount BEFORE checking balance. Zero and negative must be caught first.

import java.util.*;

class LoanAccount {
    private long   loanAccountNumber;
    private String accountHolderName;
    private double loanAmount;
    private String bankName;
    private double interestRate;

    public LoanAccount(long loanAccountNumber, String accountHolderName,
                       double loanAmount, String bankName, double interestRate) {
        this.loanAccountNumber = loanAccountNumber;
        this.accountHolderName = accountHolderName;
        this.loanAmount = loanAmount;
        this.bankName = bankName;
        this.interestRate = interestRate;
    }

    public long   getLoanAccountNumber()  { return loanAccountNumber; }
    public String getAccountHolderName() { return accountHolderName; }
    public double getLoanAmount()        { return loanAmount; }
    public String getBankName()          { return bankName; }
    public double getInterestRate()      { return interestRate; }

    public void setLoanAccountNumber(long n)   { loanAccountNumber = n; }
    public void setAccountHolderName(String n) { accountHolderName = n; }
    public void setLoanAmount(double a)        { loanAmount = a; }
    public void setBankName(String b)          { bankName = b; }
    public void setInterestRate(double r)      { interestRate = r; }
}

public class Solution {

    public static List<LoanAccount> filterLowInterestAccounts(
            List<LoanAccount> accounts, double threshold) {
        List<LoanAccount> result = new ArrayList<>();
        for (LoanAccount acc : accounts) {
            if (acc.getInterestRate() < threshold) result.add(acc);
        }
        return result;
    }

    public static LoanAccount getHighestLoanAmountAccount(
            List<LoanAccount> accounts, String bankName) {
        LoanAccount highest = null;
        for (LoanAccount acc : accounts) {
            if (acc.getBankName().equalsIgnoreCase(bankName)) {
                if (highest == null || acc.getLoanAmount() > highest.getLoanAmount())
                    highest = acc;
            }
        }
        return highest;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(); sc.nextLine();
        List<LoanAccount> accounts = new ArrayList<>();

        for (int i = 0; i < n; i++) {
            long   accNum = sc.nextLong();  sc.nextLine();
            String name   = sc.nextLine();
            double amount = sc.nextDouble(); sc.nextLine();
            String bank   = sc.nextLine();
            double rate   = sc.nextDouble(); sc.nextLine();
            accounts.add(new LoanAccount(accNum, name, amount, bank, rate));
        }

        double threshold = sc.nextDouble(); sc.nextLine();
        String bankName  = sc.nextLine();

        List<LoanAccount> filtered = filterLowInterestAccounts(accounts, threshold);
        if (!filtered.isEmpty()) {
            for (LoanAccount acc : filtered) {
                System.out.println(acc.getLoanAccountNumber());
                System.out.println(acc.getAccountHolderName());
                System.out.println(acc.getLoanAmount());
                System.out.println(acc.getBankName());
                System.out.println(acc.getInterestRate());
            }
        } else {
            System.out.println("No accounts found with interest rate lower than the given threshold.");
        }

        LoanAccount highest = getHighestLoanAmountAccount(accounts, bankName);
        if (highest != null) {
            System.out.println(highest.getLoanAccountNumber());
            System.out.println(highest.getAccountHolderName());
            System.out.println(highest.getLoanAmount());
            System.out.println(highest.getBankName());
            System.out.println(highest.getInterestRate());
        } else {
            System.out.println("No highest loan amount account found for the mentioned bank.");
        }
        sc.close();
    }
}

  • Filter by bankName (equalsIgnoreCase)

import java.util.*;

class TrainTicket {
    private int    ticketId;
    private String travellerName, source, destination, tierClass;
    private double price;

    public TrainTicket(int ticketId, String travellerName, String source,
                       String destination, double price, String tierClass) {
        this.ticketId = ticketId; this.travellerName = travellerName;
        this.source = source;     this.destination = destination;
        this.price = price;        this.tierClass = tierClass;
    }

    public int    getTicketId()        { return ticketId; }
    public String getTravellerName()  { return travellerName; }
    public String getSource()         { return source; }
    public String getDestination()    { return destination; }
    public double getPrice()           { return price; }
    public String getTierClass()      { return tierClass; }

    public void setTicketId(int id)              { ticketId = id; }
    public void setTravellerName(String n)       { travellerName = n; }
    public void setSource(String s)              { source = s; }
    public void setDestination(String d)         { destination = d; }
    public void setPrice(double p)               { price = p; }
    public void setTierClass(String t)           { tierClass = t; }
}

public class Solution {

    public static TrainTicket findTicketWithMinPriceForGivenSource(
            TrainTicket[] tickets, String source) {
        TrainTicket min = null;
        for (TrainTicket t : tickets) {
            if (t.getSource().equalsIgnoreCase(source)) {
                if (min == null || t.getPrice() < min.getPrice()) min = t;
            }
        }
        return min;
    }

    public static double findAveragePriceOfTicketsForTierClass(
            TrainTicket[] tickets, String tierClass) {
        double total = 0; int count = 0;
        for (TrainTicket t : tickets) {
            if (t.getTierClass().equalsIgnoreCase(tierClass)) { total += t.getPrice(); count++; }
        }
        return (count == 0) ? 0 : total / count;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(); sc.nextLine();
        TrainTicket[] tickets = new TrainTicket[n];
        for (int i = 0; i < n; i++) {
            int    id   = sc.nextInt();    sc.nextLine();
            String name = sc.nextLine();
            String src  = sc.nextLine();
            String dest = sc.nextLine();
            double price= sc.nextDouble(); sc.nextLine();
            String tier = sc.nextLine();
            tickets[i] = new TrainTicket(id, name, src, dest, price, tier);
        }
        String srcP  = sc.nextLine();
        String tierP = sc.nextLine();

        TrainTicket minT = findTicketWithMinPriceForGivenSource(tickets, srcP);
        if (minT != null) { System.out.println(minT.getTicketId()); System.out.println(minT.getSource()); System.out.println(minT.getPrice()); }
        else System.out.println("No such Ticket found with mentioned attribute");

        double avg = findAveragePriceOfTicketsForTierClass(tickets, tierP);
        if (avg != 0) System.out.println(avg);
        else System.out.println("No such tier class found with mentioned attribute");

        sc.close();
    }
}

Consumer Billing System

import java.util.*;

class Consumer {
    private String consumerId, consumerName;
    private int    unitsConsumed;
    private double billAmount;

    public Consumer(String id, String name, int units) {
        consumerId = id; consumerName = name; unitsConsumed = units;
    }

    public String getConsumerId()   { return consumerId; }
    public String getConsumerName() { return consumerName; }
    public int    getUnitsConsumed() { return unitsConsumed; }
    public double getBillAmount()   { return billAmount; }
    public void   setBillAmount(double b) { billAmount = b; }
}

public class Solution {

    public static void calculateBill(ArrayList<Consumer> consumers) {
        if (consumers.isEmpty()) { System.out.println("No consumers present in the list"); return; }
        for (Consumer c : consumers) {
            int    units = c.getUnitsConsumed();
            double bill;

            if      (units <= 100) bill = units * 3;
            else if (units <= 300) bill = 100*3 + (units-100)*5;
            else if (units <= 500) bill = 100*3 + 200*5 + (units-300)*8;
            else                  bill = 100*3 + 200*5 + 200*8 + (units-500)*10;

            if (units > 300) bill *= 1.05;   // 5% surcharge
            if (bill  > 3000) bill *= 0.90;  // 10% discount
            bill += 50;                        // fixed charge

            c.setBillAmount(bill);
            System.out.println(c.getConsumerName());
            System.out.println(c.getBillAmount());
        }
    }

    public static void findPersonDetails(String name, ArrayList<Consumer> consumers) {
        for (Consumer c : consumers) {
            if (c.getConsumerName().equalsIgnoreCase(name)) {
                System.out.println(c.getConsumerId());
                System.out.println(c.getConsumerName());
                System.out.println(c.getUnitsConsumed());
                System.out.println(c.getBillAmount());
                return;
            }
        }
        System.out.println("No such consumer found");
    }

    public static void main(String[] args) throws Exception {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(); sc.nextLine();
        ArrayList<Consumer> list = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            String id    = sc.nextLine();
            String name  = sc.nextLine();
            int    units = sc.nextInt(); sc.nextLine();
            list.add(new Consumer(id, name, units));
        }
        String search = sc.nextLine();
        calculateBill(list);
        findPersonDetails(search, list);
        sc.close();
    }
}

Billing Slabs

  • 1–100 units → ₹3/unit
  • 101–300 units → ₹5/unit
  • 301–500 units → ₹8/unit
  • >500 units → ₹10/unit

Adjustments (in order)

  • If units > 300 → bill × 1.05 (surcharge)
  • If bill > 3000 → bill × 0.90 (discount)
  • Always → bill + 50 (fixed charge)

import java.io.*;

class Result {
    public static boolean checkAnagram(String s1, String s2) {
        if (s1.length() != s2.length()) return false;

        int[] freq = new int[26];
        for (char c : s1.toCharArray()) freq[c - 'a']++;
        for (char c : s2.toCharArray()) freq[c - 'a']--;

        for (int f : freq) if (f != 0) return false;
        return true;
    }
}

public class Solution {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String s1 = br.readLine().trim();
        String s2 = br.readLine().trim();
        System.out.println(Result.checkAnagram(s1, s2));
    }
}

Student Ranking — Java

import java.util.*;

class Student {
    private int    id, marks, age;
    private String name;

    public Student(int id, String name, int marks, int age) {
        this.id = id; this.name = name; this.marks = marks; this.age = age;
    }

    public int    getId()    { return id; }
    public String getName() { return name; }
    public int    getMarks() { return marks; }
    public int    getAge()   { return age; }

    public void setId(int id)        { this.id = id; }
    public void setName(String n)    { name = n; }
    public void setMarks(int m)     { marks = m; }
    public void setAge(int a)       { age = a; }
}

public class Solution {

    public static Student findStudentWithMaxAgeMinMarks(Student[] students) {
        if (students == null || students.length == 0) return null;
        Student res = students[0];
        for (Student s : students) {
            if (s.getAge() > res.getAge() ||
               (s.getAge() == res.getAge() && s.getMarks() < res.getMarks())) res = s;
        }
        return res;
    }

    public static Student[] searchStudentByName(Student[] students, String name) {
        List<Student> matched = new ArrayList<>();
        for (Student s : students)
            if (s.getName().equalsIgnoreCase(name)) matched.add(s);
        if (matched.isEmpty()) return null;
        matched.sort((a, b) -> b.getMarks() - a.getMarks());
        return matched.toArray(new Student[0]);
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(); sc.nextLine();
        Student[] students = new Student[n];
        for (int i = 0; i < n; i++) {
            int    id   = sc.nextInt();  sc.nextLine();
            String name = sc.nextLine();
            int    marks= sc.nextInt();
            int    age  = sc.nextInt();  sc.nextLine();
            students[i] = new Student(id, name, marks, age);
        }
        String searchName = sc.nextLine();

        Student maxAge = findStudentWithMaxAgeMinMarks(students);
        if (maxAge == null) System.out.println("No Students found");
        else { System.out.println("id-"+maxAge.getId()); System.out.println("name-"+maxAge.getName()); System.out.println("marks-"+maxAge.getMarks()); System.out.println("age-"+maxAge.getAge()); }

        Student[] found = searchStudentByName(students, searchName);
        if (found == null) System.out.println("No Student found with mentioned attribute.");
        else for (Student s : found) { System.out.println("id-"+s.getId()); System.out.println("name-"+s.getName()); System.out.println("marks-"+s.getMarks()); System.out.println("age-"+s.getAge()); }

        sc.close();
    }
}

Neon Number — Exception Handling

Exception Handling Medium

import java.util.*;

class NotNeonException extends Exception {
    public NotNeonException(String message) { super(message); }
}

public class Solution {

    static boolean isNeon(int n) {
        if (n == 0) return true;   // 0² = 0, digit sum = 0 == 0
        long square = (long) n * n;
        int digitSum = 0;
        while (square > 0) { digitSum += square % 10; square /= 10; }
        return digitSum == n;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int size = sc.nextInt();
        List<Integer> numbers = new ArrayList<>();
        for (int i = 0; i < size; i++) numbers.add(sc.nextInt());

        for (int num : numbers) {
            try {
                if (isNeon(num)) System.out.println(num);
                else throw new NotNeonException(num + " is not a neon number");
            } catch (NotNeonException e) {
                System.out.println(e.getClass().getSimpleName() + ": " + e.getMessage());
            }
        }
        sc.close();
    }
}

What is a Neon Number?

  • n=0 → 0²=0, sum=0 → ✓
  • n=1 → 1²=1, sum=1 → ✓
  • n=9 → 9²=81, 8+1=9 → ✓
  • n=8 → 8²=64, 6+4=10 ≠ 8 → ✗
  • n=29 → 29²=841, 8+4+1=13 ≠ 29 → ✗

Output format

If neon → print the number.
If not → print NotNeonException: <n> is not a neon number
