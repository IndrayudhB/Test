import java.util.*;

class Result {
    public static int getSessionCount(int timeout,
            List<String> userIds, List<Integer> timestamps) {
        Map<String, List<Integer>> map = new HashMap<>();
        for (int i = 0; i < userIds.size(); i++) {
            map.putIfAbsent(userIds.get(i), new ArrayList<>());
            map.get(userIds.get(i)).add(timestamps.get(i));
        }
        int totalSessions = 0;
        for (String user : map.keySet()) {
            List<Integer> times = map.get(user);
            Collections.sort(times);
            int sessions = 1;
            for (int i = 1; i < times.size(); i++)
                if (times.get(i) - times.get(i-1) > timeout) sessions++;
            totalSessions += sessions;
        }
        return totalSessions;
    }
}
from collections import defaultdict

def getSessionCount(timeout, userIds, timestamps):
    user_map = defaultdict(list)
    for uid, ts in zip(userIds, timestamps):
        user_map[uid].append(ts)
    total = 0
    for uid, times in user_map.items():
        times.sort()
        sessions = 1
        for i in range(1, len(times)):
            if times[i] - times[i-1] > timeout:
                sessions += 1
        total += sessions
    return total

import java.util.*;

class Result {
    public static int getRetryCount(int gap,
            List<String> requestIds, List<Integer> timestamps) {
        Map<String, Integer> lastSeen = new HashMap<>();
        int retryCount = 0;
        for (int i = 0; i < requestIds.size(); i++) {
            String req = requestIds.get(i);
            int time = timestamps.get(i);
            if (lastSeen.containsKey(req) && time - lastSeen.get(req) <= gap)
                retryCount++;
            lastSeen.put(req, time);
        }
        return retryCount;
    }
}
def getRetryCount(gap, requestIds, timestamps):
    last_seen = {}
    retry_count = 0
    for req, time in zip(requestIds, timestamps):
        if req in last_seen and time - last_seen[req] <= gap:
            retry_count += 1
        last_seen[req] = time
    return retry_count

import java.util.*;

class Result {
    public static long visibilityAdjustment(List<List<Integer>> visibilityScore) {
        int n = visibilityScore.size(), m = visibilityScore.get(0).size();
        long cost = 0;
        for (int j = 0; j < m; j++) {
            int prev = visibilityScore.get(0).get(j);
            for (int i = 1; i < n; i++) {
                int cur = visibilityScore.get(i).get(j);
                int req = prev + 1;
                if (cur < req) { cost += (req - cur); cur = req; }
                prev = cur;
            }
        }
        return cost;
    }
}
def visibilityAdjustment(visibilityScore):
    n, m = len(visibilityScore), len(visibilityScore[0])
    cost = 0
    for j in range(m):
        prev = visibilityScore[0][j]
        for i in range(1, n):
            cur = visibilityScore[i][j]
            req = prev + 1
            if cur < req:
                cost += (req - cur)
                cur = req
            prev = cur
    return cost

import java.util.*;

class Result {
    public static int groupDivision(List<Integer> levels, int maxSpread) {
        Collections.sort(levels);
        int groups = 1, start = levels.get(0);
        for (int i = 1; i < levels.size(); i++) {
            if (levels.get(i) - start > maxSpread) {
                groups++;
                start = levels.get(i);
            }
        }
        return groups;
    }
}
def groupDivision(levels, maxSpread):
    levels.sort()
    groups, start = 1, levels[0]
    for lvl in levels[1:]:
        if lvl - start > maxSpread:
            groups += 1
            start = lvl
    return groups

FROM companies c
JOIN campaigns cp ON c.id = cp.company_id

import java.net.*;
import java.io.*;
import org.json.*;

class Result {
    public static String bestInGenre(String genre) throws Exception {
        JSONObject first = fetchJson("https://jsonmock.hackerrank.com/api/tvseries?page=1");
        int totalPages = first.getInt("total_pages");
        String best = null;
        double bestRating = -1;
        for (int page = 1; page <= totalPages; page++) {
            JSONArray data = fetchJson(
                "https://jsonmock.hackerrank.com/api/tvseries?page=" + page
            ).getJSONArray("data");
            for (int i = 0; i < data.length(); i++) {
                JSONObject show = data.getJSONObject(i);
                if (show.getString("genre").contains(genre)) {
                    double rating = show.getDouble("imdb_rating");
                    String name = show.getString("name");
                    if (rating > bestRating ||
                        (rating == bestRating && name.compareTo(best) < 0)) {
                        bestRating = rating; best = name;
                    }
                }
            }
        }
        return best != null ? best : "";
    }

    private static JSONObject fetchJson(String url) throws Exception {
        HttpURLConnection c = (HttpURLConnection) new URL(url).openConnection();
        c.setRequestMethod("GET");
        BufferedReader br = new BufferedReader(new InputStreamReader(c.getInputStream()));
        StringBuilder sb = new StringBuilder(); String ln;
        while ((ln = br.readLine()) != null) sb.append(ln);
        return new JSONObject(sb.toString());
    }
}
import requests, os

def bestInGenre(genre):
    r = requests.get("https://jsonmock.hackerrank.com/api/tvseries?page=1").json()
    total_pages = r['total_pages']
    best, best_rating = None, -1
    for page in range(1, total_pages + 1):
        data = requests.get(
            f"https://jsonmock.hackerrank.com/api/tvseries?page={page}"
        ).json()['data']
        for show in data:
            if genre in show['genre']:
                rating, name = show['imdb_rating'], show['name']
                if rating > best_rating or (rating == best_rating and name < best):
                    best_rating, best = rating, name
    return best if best else ""

if __name__ == '__main__':
    fptr = open(os.environ['OUTPUT_PATH'], 'w')

    public static String mostSuccessfulDomesticClub(String league, double weight)
            throws Exception {

            }
            if (page >= r.getInt("total_pages")) break;
            page++;

    public static long getDiscountedPrice(long barcode) throws Exception {

            "https://jsonmock.hackerrank.com/api/inventory?barcode=" + barcode);
